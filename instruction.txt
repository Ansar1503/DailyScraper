on linux : 
# 1. Create a new virtual environment
python3 -m venv .venv

# 2. Activate it
source .venv/bin/activate

# 3. Install the dependencies
pip install -r requirement.txt

# 4. Run the scraper
python scraper.py


on windows :
# 1. Create a new virtual environment
python -m venv .venv

# 2. Activate it
.venv\Scripts\Activate.ps1

# 3. Install the dependencies
pip install -r requirement.txt

# 4. Run the scraper
python scraper.py
